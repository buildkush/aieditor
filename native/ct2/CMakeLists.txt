cmake_minimum_required(VERSION 3.20)
project(aiditor_ct2 LANGUAGES CXX C)
include(FetchContent)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
set(BUILD_CLI OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WITH_CUDA OFF CACHE BOOL "" FORCE)
set(WITH_CUDNN OFF CACHE BOOL "" FORCE)
set(ENABLE_CPU_DISPATCH ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
if(NOT DEFINED AIDITOR_CT2_BACKEND)
  message(FATAL_ERROR "Choose a CPU backend: -DAIDITOR_CT2_BACKEND=mkl|dnnl|ruy|accelerate")
endif()
foreach(backend MKL DNNL RUY ACCELERATE)
  string(TOLOWER "${backend}" lower)
  if(lower STREQUAL AIDITOR_CT2_BACKEND)
    set(WITH_${backend} ON CACHE BOOL "" FORCE)
  else()
    set(WITH_${backend} OFF CACHE BOOL "" FORCE)
  endif()
endforeach()
FetchContent_Declare(ctranslate2
  GIT_REPOSITORY https://github.com/OpenNMT/CTranslate2.git
  GIT_TAG 5de572ef60b67a0276c5989ee73d0072808ff4c3
  GIT_SUBMODULES third_party/cpu_features third_party/spdlog third_party/ruy
  GIT_SUBMODULES_RECURSE TRUE)
FetchContent_GetProperties(ctranslate2)
if(NOT ctranslate2_POPULATED)
  FetchContent_Populate(ctranslate2)
endif()
include(${CMAKE_CURRENT_LIST_DIR}/patch-ct2.cmake)
add_subdirectory(${ctranslate2_SOURCE_DIR} ${ctranslate2_BINARY_DIR} EXCLUDE_FROM_ALL)
add_library(aiditor_ct2 SHARED bridge.cpp)
target_link_libraries(aiditor_ct2 PRIVATE ctranslate2)
target_include_directories(aiditor_ct2 PRIVATE ${ctranslate2_SOURCE_DIR}/include ${ctranslate2_SOURCE_DIR}/third_party)
target_compile_definitions(aiditor_ct2 PRIVATE AIDITOR_CT2_BACKEND="${AIDITOR_CT2_BACKEND}")
add_executable(aiditor_ct2_probe probe.cpp)
target_link_libraries(aiditor_ct2_probe PRIVATE ctranslate2)
enable_testing()
add_test(NAME tara_compiled_context COMMAND aiditor_ct2_probe)
if(APPLE)
  set_target_properties(aiditor_ct2 PROPERTIES INSTALL_RPATH "@loader_path")
elseif(UNIX)
  set_target_properties(aiditor_ct2 PROPERTIES INSTALL_RPATH "$ORIGIN")
endif()
install(TARGETS aiditor_ct2 ctranslate2 RUNTIME DESTINATION . LIBRARY DESTINATION .)
